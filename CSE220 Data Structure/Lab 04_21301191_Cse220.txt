class Node:
  def __init__(self, e, n, p):
    self.element = e
    self.next = n
    self.prev = p


class DoublyList:
  
  def __init__(self, a):
  #  Design the constructor based on data type of a. If 'a' is built in python list then
  #  Creates a linked list using the values from the given array.
    self.head = None
    # To Do
    self.head=Node(None,None,None)
    if type(a)==list:
      tail=self.head
      for i in range(0,len(a)):
       n=Node(a[i],None,None)
       tail.next=n
       n.prev=tail
       n.next=self.head
       self.head.prev=n
       tail=tail.next

  
  # Counts the number of Nodes in the list
  def countNode(self):
    # To Do
    temp=self.head.next
    c=0
    while (temp is not self.head):
      c=c+1
      temp=temp.next
    return c

  
  # prints the elements in the list
  def forwardprint(self):
    # To Do
    temp=self.head.next
    while temp!=self.head:
      if temp.next!=self.head:
        print(temp.element,end=",")
      else:
         print(temp.element)
      temp=temp.next


  # prints the elements in the list backward
  def backwardprint(self):
    # To Do
     tail=self.head.prev
     while tail!=self.head:
      if tail.prev!=self.head:
        print(tail.element,end=",")
      else:
        print(tail.element)
      tail=tail.prev
    

  # returns the reference of the at the given index. For invalid index return None.
  def nodeAt(self, idx):
    # To Do
    b=self.head.next
    count=0
    while (b is not self.head):
      if count==idx:
        return b
      count+=1
      b=b.next
    print("index error")
  
  
  # returns the index of the containing the given element. if the element does not exist in the List, return -1.
  def indexOf(self, elem):
    # To Do
    temp=self.head.next
    c=0 
    while temp!=self.head:
      if temp.element==elem:
        return c
      c=c+1
      temp=temp.next
    return -1


  # inserts containing the given element at the given index Check validity of index. 
  def insert(self, elem, idx):
    # To Do
    if(idx<0 and idx<=self.countNode()):
        print("Invalid index")
    newNode = Node(elem, None,None)
    second= self.head.next
    if (idx==0):
        newNode.next = second
        newNode.prev=self.head
        self.head.next = newNode
        second.prev=newNode
    elif (idx==self.countNode()):
        var=self.nodeAt(idx-1)
        newvar=newNode
        var.next=newNode
        newNode.prev=var
        newNode.next=self.head
        self.head.prev=newNode
    else:
        first = self.nodeAt(idx-1)
        last= self.nodeAt(idx)
        newNode.next = last
        newNode.prev=first
        first.next = newNode
        last.prev=newNode



  # removes at the given index. returns element of the removed node. Check validity of index. return None if index is invalid.
  def remove(self, idx):
    # To Do
    if(idx<0 and idx<=self.countNode()):
       print("Invalid index")
    removedNode=self.nodeAt(idx)
    if (idx==0):
        temp=self.nodeAt(idx+1)
        self.head.next=temp
        temp.prev=self.head
    elif (idx==self.countNode()):
        removedNode.prev.next=self.head
        self.head.prev=removedNode.prev
    else:
        before = self.nodeAt(idx-1)
        before.next=removedNode.next
        removedNode.next.prev=before
    var= str(removedNode.element)
    removedNode.element = None
    removedNode.next = None
    return var
