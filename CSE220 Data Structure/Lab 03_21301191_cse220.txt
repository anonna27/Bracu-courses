  def __init__(self, e, n):
    self.element = e
    self.next = n

  # Count the number of nodes in the list
  def countNode(self):
    # To Do
    temp=self.head
    c=0
    while (temp is not None):
      c=c+1
      temp=temp.next
    return c
   
  
  # Print elements in the list
  def printList(self):
    # To Do
    n=self.head
    while (n is not None):
      print(n.element,end=',')
      n=n.next
    print("")
  

  # returns the reference of the Node at the given index. For invalid index return None.
  def nodeAt(self, idx):
    # To Do
    b=self.head
    count=0
    while (b is not None):
      if count==idx:
        return b
      count+=1
      b=b.next
    return None
  
  
  # returns the element of the Node at the given index. For invalid idx return None.
  def get(self, idx):
    # To Do
    temp=self.head
    counter=0
    while (temp is not None):
      if counter==idx:
        return temp.element
      counter+=1
      temp=temp.next
    return None
    
  
  # updates the element of the Node at the given index. 
  # Returns the old element that was replaced. For invalid index return None.
  # parameter: index, element
  def set(self, idx, elem):
    # To Do   
    c=0
    n=self.head
    while n!= None:
      if c==idx:
         return n.element
         n.element=elem
         
      c=c + 1
      n=n.next

   

  # returns the index of the Node containing the given element.
  # if the element does not exist in the List, return -1.
  def indexOf(self, elem):
    # To Do
    num = 0
    n = self.head
    while n!=None:
        if n.element==elem:
            return num
        num = num+ 1
        n = n.next
    return -1

  
  # returns true if the element exists in the List, return false otherwise.
  def contains(self, elem):
    # To Do
    temp=self.head
    check= False
    while temp!=None:
      if temp.element==elem:
        check=True
        break
      temp=temp.next
    return check

  

  # Makes a duplicate copy of the given List. Returns the reference of the duplicate list.
  def copyList(self):
    # To Do
    copyHead = None
    copyTail = None
    temp= self.head
    while(temp is not None):
        newNode = Node(temp.element, None)
        if copyHead == None:
            copyHead = newNode
            copyTail = newNode
        else:
            copyTail.next = newNode
            copyTail = newNode
        temp= temp.next
    return copyHead


  # Makes a reversed copy of the given List. Returns the head reference of the reversed list.
  def reverseList(self):
    # To Do
    prev = None
    current = self.head
    while current is not None:
       next = current.next
       current.next = prev
       prev = current
       current = next

    self.head = prev
    return self.head
    
  
  # inserts Node containing the given element at the given index
  # Check validity of index.
  def insert(self, elem, idx):
    # To Do
    if(idx<0 and idx<=self.countNode()):
        print("Invalid index")
    newNode = Node(elem, None)
    if (idx==0):
        newNode.next = self.head
        self.head = newNode
    else:
        pred = self.nodeAt(idx-1)
        newNode.next = pred.next
        pred.next = newNode
    return self.head



  # removes Node at the given index. returns element of the removed node.
  # Check validity of index. return None if index is invalid.
  def remove(self, idx):
    # To Do
    if(idx<0 and idx<=self.countNode()):
       print("Invalid index")
    removedNode = None
    if (idx==0):
        removedNode = self.head
        self.head = removedNode.next
    else:
        pred = self.nodeAt(idx-1)
        removedNode = pred.next
        pred.next = removedNode.next
    removedNode.element = None
    removedNode.next = None
    return self.head
  
  
  # Rotates the list to the left by 1 position.
  def rotateLeft(self):
    # To Do
    oldHead = self.head
    tail=self.head
    self.head = tail.next
    while(tail.next is not None):
        tail = tail.next
    tail.next = oldHead
    oldHead.next = None
    return self.head
 
  
  
  # Rotates the list to the right by 1 position.
  def rotateRight(self):
    # To Do
    oldHead = self.head
    tail=self.head
    num=self.head 
    while(tail.next is not None):
        tail = tail.next
        while(num.next is not tail):
          num=num.next
    self.head=tail
    tail.next = oldHead
    num.next = None
    return self.head
