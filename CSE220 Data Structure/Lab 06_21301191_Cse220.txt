class Hashtable: 
  def __init__(self,arr,size=9):
    self.size=size
    self.arr=arr
    self.hash_table=[0]*size
 
  def hashing(self,string):
    digit_sum=0
    cons_num=0
    for char in string:
        if char not in 'AEIOU0123456789':
            cons_num += 1
        elif char in '0123456789':
            digit_sum += int(char) 
    store_val=(cons_num*24 + digit_sum) % 9 
    return store_val


  def linear_probing(self):
    for i in range(len(self.arr)):
      if i>=9:
        break
      else:
       index = self.hashing(self.arr[i])  
       while self.hash_table[index]!= 0:
          index=(index+1)%len(self.hash_table)
       self.hash_table[index]=self.arr[i]
    return self.hash_table
