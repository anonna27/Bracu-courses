    self.start = 0

  # This method will sort the values by keeping the start unchanged
  def sort(self):
    # To Do
    temp=0
    for z in range(0,self.sz): 
        for y in range(z+1,self.sz):
          if self.cir[(y+self.st)%len(self.cir)]<self.cir[(z+self.st)%len(self.cir)]:
            temp=self.cir[(z+self.st)%len(self.cir)]
            self.cir[(z+self.st)%len(self.cir)]=self.cir[(y+self.st)%len(self.cir)]
            self.cir[(y+self.st)%len(self.cir)]=temp
    return self.cir

  def equivalent(self, cir_arr):

    start2=cir_arr.st
    for i in range(0,self.sz):
