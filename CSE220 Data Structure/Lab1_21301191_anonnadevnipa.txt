def shiftLeft(source, k):
  for i in range(0,k):
    for j in range(0,len(source)-1):
      source[j]=source[j+1]
    source[len(source)-1] =0
  return source

def shiftRight(source,k):
  for j in range(0,k):
   for i in range(len(source)-1,0,-1):
    source[i]=source[i-1]
   source[j]=0
  return source

#Task 4
def rotateRight(source,k):
  for i in range(0,k):
    index= source[len(source)-1]
    for j in range(len(source)-1,0,-1):
      source[j]=source[j-1]
    source[0]=index
  return source

def remove(source,size,index):
    for i in range(index+1,size):
      source[i-1]=source[i]
    source[size-1]= 0
    return source

def series(n):
  arr=[0]*(n*n)
  segment=1
  x=0
  for i in range(len(arr)):
    if x >= n-segment:
      arr[i]= n-x
    x+=1
    if x==n:
